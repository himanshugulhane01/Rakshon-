  <script>
    function updateClock() {
      const now = new Date();
      let hours = now.getHours();
      let minutes = now.getMinutes();
      minutes = minutes < 10 ? '0' + minutes : minutes;
      document.getElementById('clockDisplay').textContent = `${hours}:${minutes}`;
    }
    updateClock();
    setInterval(updateClock, 30000);

    const shieldContainer = document.getElementById('shieldContainer');
    shieldContainer.addEventListener('click', () => {
      shieldContainer.style.transform = 'scale(1.16)';
      burstCyberParticles(40);
      setTimeout(() => {
        shieldContainer.style.transform = '';
      }, 240);
    });

    const progressFill = document.getElementById('progressFill');
    const statusLabel = document.getElementById('statusLabel');
    const btnExplore = document.getElementById('btnExplore');

    let currentProgress = 0;
    let progressTimer = null;

    function runProgress() {
      currentProgress = 0;
      progressFill.style.width = '0%';
      statusLabel.textContent = 'Securing your digital world...';
      btnExplore.style.display = 'none';

      clearInterval(progressTimer);
      progressTimer = setInterval(() => {
        if (currentProgress < 68) {
          currentProgress += 2.2;
        } else if (currentProgress < 95) {
          currentProgress += 1.1;
          statusLabel.textContent = 'Verifying threat databases...';
        } else if (currentProgress < 100) {
          currentProgress += 0.8;
        } else {
          currentProgress = 100;
          clearInterval(progressTimer);
          statusLabel.textContent = 'Digital Protection Active';
          burstCyberParticles(25);
          btnExplore.style.display = 'block';
        }
        progressFill.style.width = `${currentProgress}%`;
      }, 40);
    }

    function restartAnimation() {
      runProgress();
    }

    window.addEventListener('load', () => {
      runProgress();
    });

    const canvas = document.getElementById('particleCanvas');
    const ctx = canvas.getContext('2d');
    let particles = [];

    function resizeCanvas() {
      canvas.width = canvas.parentElement.clientWidth;
      canvas.height = canvas.parentElement.clientHeight;
    }
    window.addEventListener('resize', resizeCanvas);
    resizeCanvas();

    class CyberParticle {
      constructor(x, y, vx, vy, color, size, life) {
        this.x = x || Math.random() * canvas.width;
        this.y = y || canvas.height + 10;
        this.vx = vx || (Math.random() - 0.5) * 0.8;
        this.vy = vy || -(Math.random() * 1.4 + 0.3);
        this.color = color || (Math.random() > 0.4 ? '#38BDF8' : '#2DD4BF');
        this.size = size || (Math.random() * 2 + 1);
        this.life = life || 140;
        this.maxLife = this.life;
      }

      update() {
        this.x += this.vx;
        this.y += this.vy;
        this.life -= 1;
      }

      draw() {
        const alpha = Math.max(0, this.life / this.maxLife) * 0.5;
        ctx.save();
        ctx.globalAlpha = alpha;
        ctx.fillStyle = this.color;
        ctx.shadowColor = this.color;
        ctx.shadowBlur = 4;
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
        ctx.fill();
        ctx.restore();
      }
    }

    function burstCyberParticles(count = 30) {
      const centerX = canvas.width / 2;
      const centerY = canvas.height * 0.34;
      for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = Math.random() * 4 + 1.2;
        const vx = Math.cos(angle) * speed;
        const vy = Math.sin(angle) * speed;
        const color = Math.random() > 0.5 ? '#38BDF8' : (Math.random() > 0.5 ? '#34D399' : '#FFFFFF');
        particles.push(new CyberParticle(centerX, centerY, vx, vy, color, Math.random() * 2.5 + 1.5, 60));
      }
    }

    function particleLoop() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);

      if (particles.length < 28) {
        particles.push(new CyberParticle());
      }

      for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.update();
        p.draw();
        if (p.life <= 0) {
          particles.splice(i, 1);
        }
      }

      requestAnimationFrame(particleLoop);
    }
    particleLoop();
  </script>
